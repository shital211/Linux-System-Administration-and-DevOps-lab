Now Install Node Exporter on Jenkins Server

Step 1:- Download Node Exporter
      #cd /opt
	  #wget https://github.com/prometheus/node_exporter/releases/download/v1.11.1/node_exporter-1.11.1.linux-amd64.tar.gz
	  #ls
     #tar -xvzf node_exporter-1.11.1.linux-amd64.tar.gz
     #ls
     #rm -rf node_exporter-1.11.1.linux-amd64.tar.gz
	 #mv node_exporter-1.11.1.linux-amd64/ node_exporter
	 #ls
	
Step 2:- Create Node Exporter User
      #useradd --no-create-home --shell /bin/false node_exporter
	  
Step 3:- Change Ownership
      #chown -R node_exporter:node_exporter /opt/node_exporter
	  
Step 4:- Create Systemd Service
     #vi /etc/systemd/system/node_exporter.service
[Unit]
Description=Prometheus Node Exporter
After=network.target

[Service]
User=node_exporter
Group=node_exporter
Type=simple
ExecStart=/opt/node_exporter/node_exporter

[Install]
WantedBy=multi-user.target

:wq! 

Step 5:- Reload Systemd
      #systemctl daemon-reload
	  
Step 6:- Enable Service
      #systemctl enable --now node_exporter
	  
Step 7:- Check Status
      #systemctl status node_exporter
	  
Step 8:- Open Firewall
      #firewall-cmd --permanent --add-port=9100/tcp
       #firewall-cmd --reload
      #firewall-cmd --list-all

        #chown -R apache_exporter:apache_exporter /opt/apache_exporter
		
Step 5:- Create Systemd Service
        #vi /etc/systemd/system/apache_exporter.service
	[Unit]
Description=Prometheus Apache Exporter
After=network.target

[Service]
User=apache_exporter
Group=apache_exporter
Type=simple

ExecStart=/opt/apache_exporter/apache_exporter \
  --scrape_uri=http://localhost/server-status?auto

Restart=always

[Install]
WantedBy=multi-user.target

:wq!
