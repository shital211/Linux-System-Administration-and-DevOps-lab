---
- name: Deploy LAMP Stack on Red Hat Enterprise Linux 9.8
  hosts: web
  become: yes

  tasks:
    # Install Apache
    - name: Install Apache
      ansible.builtin.yum:
        name: httpd
        state: present

    - name: Start and enable Apache
      ansible.builtin.service:
        name: httpd
        state: started
        enabled: yes

    # Install MariaDB
    - name: Install MariaDB server
      ansible.builtin.yum:
        name: mariadb-server
        state: present

    - name: Start and enable MariaDB
      ansible.builtin.service:
        name: mariadb
        state: started
        enabled: yes

    # Install PHP
    - name: Install PHP and required modules
      ansible.builtin.yum:
        name:
          - php
          - php-mysqlnd
          - php-cli
          - php-common
          - php-gd
          - php-xml
          - php-mbstring
        state: present

    - name: Restart Apache to load PHP
      ansible.builtin.service:
        name: httpd
        state: restarted

    # Firewall configuration without ansible.posix
    - name: Allow HTTP
      command: firewall-cmd --permanent --add-service=http


    - name: Allow HTTPS
      command: firewall-cmd --permanent --add-service=https

    - name: Reload firewall
      command: firewall-cmd --reload

    # Test PHP page
    - name: Deploy a test PHP file
      ansible.builtin.copy:
        dest: /var/www/html/info.php
        content: |
          <?php
          phpinfo();
          ?>
        mode: '0644'
